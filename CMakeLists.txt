cmake_minimum_required(VERSION 3.16)
if(COMMAND idf_component_register)
    idf_component_register(SRCS "src/wifi.c" "src/ble.c" INCLUDE_DIRS "include")
    return()
endif()
project(espargos_packet_decoders VERSION 0.1.0 LANGUAGES C)
add_library(espargos_packet_decoders src/wifi.c src/ble.c)
target_include_directories(espargos_packet_decoders PUBLIC include)
target_compile_features(espargos_packet_decoders PUBLIC c_std_99)
target_link_libraries(espargos_packet_decoders PUBLIC m)
set_target_properties(espargos_packet_decoders PROPERTIES POSITION_INDEPENDENT_CODE ON)
install(TARGETS espargos_packet_decoders)
install(DIRECTORY include/ DESTINATION include)
